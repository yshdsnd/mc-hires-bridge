cmake_minimum_required(VERSION 3.20)
project(MCHiResBridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
# Preserve build/Release paths used by packaging and diagnostic commands.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
option(MC_BRIDGE_BUILD_DIAGNOSTICS "Build audio observation tools" OFF)
option(MC_BRIDGE_BUILD_EXPERIMENTS "Build research and legacy bypass targets" OFF)
add_subdirectory(vendor/minhook EXCLUDE_FROM_ALL)
if(CMAKE_SIZEOF_VOID_P EQUAL 4)
  enable_language(RC)
  add_subdirectory(src)
  add_subdirectory(tests)
  install(FILES LICENSE DESTINATION .)
  install(FILES docs/package-readme.md DESTINATION . RENAME README.md)
  install(FILES vendor/minhook/LICENSE.txt DESTINATION licenses RENAME MinHook.txt)
endif()
if(MC_BRIDGE_BUILD_DIAGNOSTICS)
  add_subdirectory(diagnostics)
endif()
if(MC_BRIDGE_BUILD_EXPERIMENTS)
  add_subdirectory(experiments)
endif()
